# Reduced dependency set for the offline demo (`make demo`) and the Docker image.
#
# WHY THIS FILE EXISTS
# --------------------
# requirements.txt pins torch, transformers, sentence-transformers, chromadb,
# scipy and scikit-learn. Together those are well over 1 GB of wheels, and the
# demo path touches none of them: the ChromaDB-backed CharacterAnalysisAgent is
# replaced by a canned stand-in (see scripts/demo.py), and both agents/ and
# utils/ already guard those imports behind try/except ImportError.
#
# This is the *demo* dependency set, not the production one. A full run
# (real Gemini summarisation, Imagen frames, ChromaDB character memory) still
# needs `pip install -r requirements.txt`.
#
# Deliberate substitutions vs requirements.txt:
#   * opencv-python-headless instead of opencv-python - identical API for the
#     imread/calcHist/kmeans calls in core/visual_coherence_manager.py, without
#     the GUI stack (libGL, libglib) that a container has no use for.
#   * langchain-google-genai is omitted - the demo patches init_chat_model, so
#     the Gemini provider package is never constructed. The langchain factory
#     itself still imports, which is all the orchestrator needs at import time.

# LangChain core (prompt templates + the init_chat_model factory the demo patches)
langchain==0.3.27
langchain-core==0.3.72
langchain-text-splitters==0.3.9

# google-genai still has to be present, but no longer for the reason this comment
# used to give. f2597bf made it an *optional* import in media/media_utils.py, so
# that module now imports fine without it and degrades to placeholder frames and
# silent audio. What still requires it is agents/video_agent.py, which does a
# bare `from google import genai` at module scope; WorkflowOrchestrator imports
# VideoGenerationAgent, so the demo cannot start without the package installed.
# The demo never builds a client: the credentials are cleared (twice - see
# scripts/demo.py::clear_credentials) and the run aborts unless
# build_image_generator() refuses to construct one.
google-genai>=0.8.0
